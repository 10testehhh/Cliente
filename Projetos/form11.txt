        // 5. Adicionamos os métodos da interface com a lógica da animação
        public void UpdateAnimation()
        {
            if (!IsAnimationValid()) return;

            Point mousePosition = Point.Empty;
            this.Invoke((Action)(() => {
                mousePosition = this.PointToClient(Cursor.Position);
            }));

            foreach (var p in particles)
            {
                p.Update(this.ClientSize.Width, this.ClientSize.Height);
                float dx = p.Position.X - mousePosition.X;
                float dy = p.Position.Y - mousePosition.Y;
                float distance = (float)Math.Sqrt(dx * dx + dy * dy);

                if (distance < MOUSE_REPEL_RADIUS && distance > 0)
                {
                    float repelForce = MOUSE_REPEL_STRENGTH * (1 - (distance / MOUSE_REPEL_RADIUS));
                    p.ApplyForce(new PointF(dx / distance * repelForce, dy / distance * repelForce));
                }
            }
        }

        public void InvalidateAnimation()
        {
            if (IsAnimationValid()) this.BeginInvoke((Action)(() => this.Invalidate(true)));
        }

        public bool IsAnimationValid()
        {
            return !this.IsDisposed && this.IsHandleCreated;
        }
