        #region Implementação da Interface IAnimatable
        // 4. Adicionamos os métodos da interface com a lógica da animação
        public void UpdateAnimation()
        {
            if (!IsAnimationValid()) return;

            Point mousePosition = Point.Empty;
            this.Invoke((Action)(() => {
                mousePosition = this.PointToClient(Cursor.Position);
            }));

            foreach (var p in particles)
            {
                p.Update(this.ClientSize.Width, this.ClientSize.Height); // Assumindo que ParticleRGB tem este método
                float dx = p.Position.X - mousePosition.X;
                float dy = p.Position.Y - mousePosition.Y;
                float distance = (float)Math.Sqrt(dx * dx + dy * dy);

                if (distance < MOUSE_REPEL_RADIUS && distance > 0)
                {
                    float repelForce = MOUSE_REPEL_STRENGTH * (1 - (distance / MOUSE_REPEL_RADIUS));
                    p.ApplyForce(new PointF(dx / distance * repelForce, dy / distance * repelForce));
                }
            }
        }

        public void InvalidateAnimation()
        {
            if (IsAnimationValid()) this.BeginInvoke((Action)(() => this.Invalidate(true)));
        }

        public bool IsAnimationValid()
        {
            return !this.IsDisposed && this.IsHandleCreated;
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);
            if (particles == null) return;
            Graphics g = e.Graphics;
            g.SmoothingMode = SmoothingMode.AntiAlias;

            using (Pen linePen = new Pen(Color.White, 1.5f))
            using (SolidBrush particleBrush = new SolidBrush(Color.White))
            {
                for (int i = 0; i < particles.Count; i++)
                {
                    for (int j = i + 1; j < particles.Count; j++)
                    {
                        ParticleRGB p1 = particles[i];
                        ParticleRGB p2 = particles[j];
                        float dx = p1.Position.X - p2.Position.X;
                        float dy = p1.Position.Y - p2.Position.Y;
                        float distance = (float)Math.Sqrt(dx * dx + dy * dy);

                        if (distance < CONNECTION_DISTANCE)
                        {
                            byte alpha = (byte)(200 * (1 - (distance / CONNECTION_DISTANCE)));
                            linePen.Color = Color.FromArgb(alpha, p1.ParticleColor);
                            g.DrawLine(linePen, p1.Position, p2.Position);
                        }
                    }
                }

                foreach (var p in particles)
                {
                    particleBrush.Color = p.ParticleColor;
                    g.FillEllipse(particleBrush, p.Position.X - p.Size / 2, p.Position.Y - p.Size / 2, p.Size, p.Size);
                }
            }
        }

        private void HideScrollBars()

        }

        private void Form13SemBorda_MouseDown(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left)
            {
                ReleaseCapture();
                SendMessage(Handle, WM_NCLBUTTONDOWN, HT_CAPTION, 0);
            }
        }
        private void SetFormRegion()
        {
            if (this.Width <= 0 || this.Height <= 0) return;
            GraphicsPath path = new GraphicsPath();
            path.AddArc(0, 0, cornerRadius * 2, cornerRadius * 2, 180, 90);
            path.AddArc(this.Width - cornerRadius * 2, 0, cornerRadius * 2, cornerRadius * 2, 270, 90);
            path.AddArc(this.Width - cornerRadius * 2, this.Height - cornerRadius * 2, cornerRadius * 2, cornerRadius * 2, 0, 90);
            path.AddArc(0, this.Height - cornerRadius * 2, cornerRadius * 2, cornerRadius * 2, 90, 90);
            path.CloseAllFigures();
            this.Region = new Region(path);
        }
