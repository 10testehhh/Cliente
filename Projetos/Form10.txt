        private int cornerRadius = 20; // O raio dos cantos. Ajuste conforme desejar.
        private List<Particle> particles;
        

        // Configurações do sistema
        private const int NUMBER_OF_PARTICLES = 20; // Quantidade de partículas
        private const float CONNECTION_DISTANCE = 100; // Distância máxima para conectar partículas
        private const float MOUSE_REPEL_RADIUS = 75; // Raio de repulsão do mouse
        private const float MOUSE_REPEL_STRENGTH = 2f; // Força de repulsão do mouse
        
        public Form10(Point location)
        {
            InitializeComponent();

            
            // Exemplo: Mostrar o status inicial (propriedade)
            
            // 1. Remova a borda padrão do formulário do Windows.
            // Isso é crucial para controlar o desenho da sua própria forma.
            this.FormBorderStyle = FormBorderStyle.None;

            // 2. Define o BackColor para a cor que você deseja que o formulário tenha.
            // Isso cobrirá qualquer área que não seja parte da sua região personalizada.
            this.BackColor = Color.LightBlue; // Exemplo: um azul claro. Mude para a cor do seu formulário.

            // 3. Adiciona manipuladores de eventos para redesenhar a região quando o formulário é redimensionado.
            this.Resize += new System.EventHandler(Form10SemBorda_Resize);

            // 4. Adiciona manipulador para permitir arrastar o formulário.
            this.MouseDown += new MouseEventHandler(Form10SemBorda_MouseDown);

            // Define a região inicial do formulário ao ser carregado.
            SetFormRegion(); // Chame no construtor para definir a região inicial
            Location = location;
            this.DoubleBuffered = true; // Evita "flicker" ao desenhar
            this.BackColor = Color.Black; // Fundo escuro para contrastar com as partículas brancas
            this.Text = "Sistema de Partículas Conectadas";

            // Inicializa as partículas
            particles = new List<Particle>();
            for (int i = 0; i < NUMBER_OF_PARTICLES; i++)
            {
                particles.Add(new Particle(this.ClientSize.Width, this.ClientSize.Height));
            }

            // Configura o timer para a animação
           
        }
        private void GameLoopTimer_Tick(object sender, EventArgs e)
        {
            // Pega a posição atual do mouse na tela do formulário
            Point mousePosition = this.PointToClient(Cursor.Position);

            // 1. Atualiza a posição de cada partícula
            foreach (var p in particles)
            {
                p.Update(this.ClientSize.Width, this.ClientSize.Height);

                // 2. Aplica repulsão se o mouse estiver perto
                float dx = p.Position.X - mousePosition.X;
                float dy = p.Position.Y - mousePosition.Y;
                float distance = (float)Math.Sqrt(dx * dx + dy * dy);

                if (distance < MOUSE_REPEL_RADIUS && distance > 0)
                {
                    // Calcula a força de repulsão inversamente proporcional à distância
                    float repelForce = MOUSE_REPEL_STRENGTH * (1 - (distance / MOUSE_REPEL_RADIUS));
                    p.ApplyForce(new PointF(dx / distance * repelForce, dy / distance * repelForce));
                }
            }

            // 3. Força o redesenho do formulário
            this.Invalidate(); // Isso chamará o evento OnPaint
        }
        private void Form10SemBorda_MouseDown(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left)
            {
                ReleaseCapture();
                SendMessage(Handle, WM_NCLBUTTONDOWN, HT_CAPTION, 0);
            }
        }
        /// <summary>
        /// Define a região do formulário com cantos arredondados.
        /// </summary>
        private void SetFormRegion()
        {
            // Crie um caminho gráfico para a forma do formulário.
            GraphicsPath path = new GraphicsPath();

            // Adicione arcos para os cantos e linhas para conectar.
            // A ordem é importante: Top-Left, Top-Right, Bottom-Right, Bottom-Left.
            path.AddArc(0, 0, cornerRadius * 2, cornerRadius * 2, 180, 90); // Canto superior esquerdo
            path.AddArc(this.Width - cornerRadius * 2, 0, cornerRadius * 2, cornerRadius * 2, 270, 90); // Canto superior direito
            path.AddArc(this.Width - cornerRadius * 2, this.Height - cornerRadius * 2, cornerRadius * 2, cornerRadius * 2, 0, 90); // Canto inferior direito
            path.AddArc(0, this.Height - cornerRadius * 2, cornerRadius * 2, cornerRadius * 2, 90, 90); // Canto inferior esquerdo

            path.CloseAllFigures(); // Fecha o caminho para formar a figura completa.

            // Aplica o caminho como a região do formulário. Isso "recorta" o formulário.
            this.Region = new Region(path);
        }
        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);

            Graphics g = e.Graphics;
            g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias; // Suaviza o desenho

            // Desenha as conexões entre as partículas
            using (Pen pen = new Pen(Color.FromArgb(50, 255, 255, 0))) // Linhas translúcidas
            {
                for (int i = 0; i < particles.Count; i++)
                {
                    for (int j = i + 1; j < particles.Count; j++)
                    {
                        Particle p1 = particles[i];
                        Particle p2 = particles[j];

                        float dx = p1.Position.X - p2.Position.X;
                        float dy = p1.Position.Y - p2.Position.Y;
                        float distance = (float)Math.Sqrt(dx * dx + dy * dy);

                        // Se a distância for menor que o limite, desenha uma linha
                        if (distance < CONNECTION_DISTANCE)
                        {
                            // Ajusta a opacidade da linha baseada na distância
                            byte alpha = (byte)(255 * (1 - (distance / CONNECTION_DISTANCE)));
                            pen.Color = Color.FromArgb(alpha, 0, 255, 0);
                            g.DrawLine(pen, p1.Position, p2.Position);
                        }
                    }
                }
            }

            // Desenha cada partícula
            using (SolidBrush brush = new SolidBrush(Color.White))
            {
                foreach (var p in particles)
                {
                    // Desenha a partícula como um círculo
                    g.FillEllipse(brush, p.Position.X - p.Size / 2, p.Position.Y - p.Size / 2, p.Size, p.Size);
                }
            }
        }
