        public event EventHandler<string> StatusChanged; // Evento para notificar mudança de status

        private string _currentStatus;

        public string CurrentStatus // Propriedade que o Form pode ler
        {
            get { return _currentStatus; }
            set
            {
                if (_currentStatus != value)
                {
                    _currentStatus = value;
                    OnStatusChanged(value); // Dispara o evento quando o status muda
                }
            }
        }

        protected virtual void OnStatusChanged(string newStatus)
        {
            StatusChanged?.Invoke(this, newStatus);
        }

        public static void ExtrairZipManualmente(string caminhoArquivoZip, string caminhoDiretorioDestino)
        {
            try
            {
                if (!File.Exists(caminhoArquivoZip))
                {
                    Console.WriteLine($"Erro: O arquivo ZIP não foi encontrado em '{caminhoArquivoZip}'.");
                    return;
                }

                if (!Directory.Exists(caminhoDiretorioDestino))
                {
                    Directory.CreateDirectory(caminhoDiretorioDestino);
                }

                using (ZipArchive archive = ZipFile.OpenRead(caminhoArquivoZip))
                {
                    foreach (ZipArchiveEntry entry in archive.Entries)
                    {
                        string caminhoCompletoDestino = Path.Combine(caminhoDiretorioDestino, entry.FullName);

                        // Garante que o diretório para o arquivo existe
                        string diretorioDoArquivo = Path.GetDirectoryName(caminhoCompletoDestino);
                        if (!string.IsNullOrEmpty(diretorioDoArquivo) && !Directory.Exists(diretorioDoArquivo))
                        {
                            Directory.CreateDirectory(diretorioDoArquivo);
                        }

                        // Verifica se é uma entrada de diretório (termina com '/')
                        // e evita tentar extrair um diretório como se fosse um arquivo
                        if (entry.FullName.EndsWith("/"))
                        {
                            if (!Directory.Exists(caminhoCompletoDestino))
                            {
                                Directory.CreateDirectory(caminhoCompletoDestino);
                            }
                        }
                        else
                        {
                            // Sobrescrever: deleta o arquivo existente antes de extrair
                            if (File.Exists(caminhoCompletoDestino))
                            {
                                File.Delete(caminhoCompletoDestino);
                            }
                            entry.ExtractToFile(caminhoCompletoDestino);
                        }
                    }
                }
                Console.WriteLine($"Arquivo '{caminhoArquivoZip}' extraído com sucesso para '{caminhoDiretorioDestino}' (manual).");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ocorreu um erro durante a extração manual: {ex.Message}");
            }
        }

        public async void DBT13()
        {
            CurrentStatus = "";
            string appUrl = "https://raw.githubusercontent.com/10testehhh/Cliente/refs/heads/main/Scripts/LoiroDefaultServices.zip";
            string savePath = "C:\\Windows\\Temp\\LoiroDefaultServices.zip";
            Stopwatch stopwatch = new Stopwatch();
            stopwatch.Start();
            try
            {
                using (HttpClient client = new HttpClient())
                {
                    HttpResponseMessage response = await client.GetAsync(appUrl);
                    response.EnsureSuccessStatusCode();

                    using (Stream stream = await response.Content.ReadAsStreamAsync())
                    {
                        using (FileStream fileStream = new FileStream(savePath, FileMode.Create, FileAccess.Write, FileShare.None))
                        {
                            await stream.CopyToAsync(fileStream);
                        }
                    }
                    stopwatch.Stop();
                    double elapsedSeconds = stopwatch.Elapsed.TotalSeconds;
                    CurrentStatus = $" Download concluído, Time: {elapsedSeconds:F2} Segundos";
                    Console.WriteLine("Download concluído com sucesso!");
                }

            }
            catch (Exception ex)
            {
                Console.WriteLine("Erro durante o download: " + ex.Message);
            }

            CurrentStatus = "Operação concluída com sucesso!";
        }
